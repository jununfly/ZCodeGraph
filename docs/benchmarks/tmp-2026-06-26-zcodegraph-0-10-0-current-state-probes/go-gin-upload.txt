1	// upload-limit-bytes/main.go
2	// Example: Restrict file upload size using Gin + http.MaxBytesReader
3	//
4	// This example shows how to use Gin together with http.MaxBytesReader to safely and *strictly*
5	// limit the maximum size of uploaded files. It returns a custom error message if the file is too large.
6	
7	package main
8	
9	import (
10		"fmt"
11		"net/http"
12	
13		"github.com/gin-gonic/gin"
14	)
15	
16	const (
17		MaxUploadSize = 1 << 20 // 1MB (can set to other sizes; used for tests and demos)
18		keyError      = "error"
19	)
20	
21	func uploadHandler(c *gin.Context) {
22		// Wrap the body reader so only MaxUploadSize bytes are allowed
23		c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, MaxUploadSize)
24	
25		// Parse multipart form (limit maxMemory for demonstration, not for size restriction)
26		if err := c.Request.ParseMultipartForm(MaxUploadSize); err != nil {
27			// Check for *http.MaxBytesError for upload size limit exceeded (portable way)
28			if _, ok := err.(*http.MaxBytesError); ok {
29				c.JSON(http.StatusRequestEntityTooLarge, gin.H{
30					keyError: fmt.Sprintf("file too large (max: %d bytes)", MaxUploadSize),
31				})
32				return
33			}
34			// Other form errors
35			c.JSON(http.StatusBadRequest, gin.H{keyError: err.Error()})
36			return
37		}
38	
39		// Get uploaded file; parameter name is "file"
40		file, _, err := c.Request.FormFile("file")
41		if err != nil {
42			c.JSON(http.StatusBadRequest, gin.H{keyError: "file form required"})
43			return
44		}
45	
46		defer func() { _ = file.Close() }()
47		// For demonstration we ignore saving the file, but could copy to disk or buffer
48		// _, err = io.Copy(io.Discard, file) // Uncomment if you want to read all bytes
49	
50		c.JSON(http.StatusOK, gin.H{
51			"message": "upload successful",
52		})
53	}
54	
55	// setupRouter creates and returns the Gin router
56	func setupRouter() *gin.Engine {
57		r := gin.Default()
58		r.POST("/upload", uploadHandler)
59		return r
60	}
61	
62	func main() {
63		r := setupRouter()
64		// Run Gin server on :8080
65		_ = r.Run(":8080")
66	}

12	type BindFile struct {

14		Email string                `form:"email" binding:"required"`
15		File  *multipart.FileHeader `form:"file"  binding:"required"`
16	}
17	
18	func main() {
19		router := gin.Default()
20		// Set a lower memory limit for multipart forms (default is 32 MiB)
21		router.MaxMultipartMemory = 8 << 20 // 8 MiB
22		router.Static("/", "./public")
23		router.POST("/upload", func(c *gin.Context) {
24			var bindFile BindFile
25	
26			// Bind file
27			if err := c.ShouldBind(&bindFile); err != nil {
28				c.String(http.StatusBadRequest, fmt.Sprintf("err: %s", err.Error()))
29				return
30			}
31	
32			// Save uploaded file
33			file := bindFile.File
34			dst := filepath.Base(file.Filename)
35			if err := c.SaveUploadedFile(file, dst); err != nil {
36				c.String(http.StatusBadRequest, fmt.Sprintf("upload file err: %s", err.Error()))
37				return
38			}
39	
40			c.String(
41				http.StatusOK,
42				fmt.Sprintf(
43					"File %s uploaded successfully with fields name=%s and email=%s.",
44					file.Filename,
45					bindFile.Name,
46					bindFile.Email,
47				),
48			)
