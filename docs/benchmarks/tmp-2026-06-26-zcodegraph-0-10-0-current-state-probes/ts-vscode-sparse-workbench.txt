12	/**
13	 * A small, bounded, persistent cache of the last-known pull-request status icon,
14	 * keyed by the pull request's link (its `https://github.com/<owner>/<repo>/pull/<n>`
15	 * URL).
16	 *
17	 * Sessions compute their PR icon from a live, asynchronously-fetched pull-request
18	 * model, so on startup the icon is missing until the first fetch completes. This
19	 * cache lets a session render its last-known icon instantly and then refine it once
20	 * the live model resolves.
21	 *
22	 * Only the {@link MAX_CACHED_ICONS} most recently updated entries are retained so
23	 * the backing storage cannot grow without bound.
24	 */
25	export interface IPullRequestIconCache {
26		readonly _serviceBrand: undefined;
27	
28		/** The last icon stored for `prLink`, or `undefined` if none is cached. */
29		get(prLink: string): ThemeIcon | undefined;
30	
31		/**
32		 * Remember `icon` as the last-known icon for `prLink`, making it the most
33		 * recently updated entry. A no-op when the icon is unchanged. Evicts the
34		 * least-recently-updated entries beyond the cap and persists the result.
35		 */
36		set(prLink: string, icon: ThemeIcon): void;
37	}

56		 * removes the first (oldest) key.
57		 */

60		constructor(

70		set(prLink: string, icon: ThemeIcon): void {
71			const existing = this._icons.get(prLink);
72			if (existing && ThemeIcon.isEqual(existing, icon)) {
73				// Icon unchanged: keep its current recency and avoid a redundant write.
74				return;
75			}
76	
77			// Re-insert so the entry becomes the most recently updated.
78			this._icons.delete(prLink);
79			this._icons.set(prLink, icon);
80	
81			// Drop the least recently updated entries beyond the cap.
82			while (this._icons.size > MAX_CACHED_ICONS) {
83				const oldest = this._icons.keys().next().value;
84				if (oldest === undefined) {
85					break;
86				}
87				this._icons.delete(oldest);
88			}
89	
90			this._save();
91		}

93		private _load(): void {
94			const raw = this._storageService.get(STORAGE_KEY, StorageScope.APPLICATION);
95			if (!raw) {
96				return;
97			}
98	
99			let entries: readonly IStoredEntry[];
100			try {
101				entries = JSON.parse(raw);
102			} catch {
103				// Corrupt cache: ignore and start fresh.
104				return;
105			}
106	
107			if (!Array.isArray(entries)) {
108				return;
109			}
110	
111			for (const entry of entries) {
112				if (entry && typeof entry.link === 'string' && ThemeIcon.isThemeIcon(entry.icon)) {
113					this._icons.set(entry.link, entry.icon);
114				}
115			}
116		}
